def caesarCipher(s,k):
    result = ''
    for i in s:
        if 65 <= ord(i) <=90:
            dif = k -(ord('Z') - ord(i))
            if dif % 26 != 0:
                result +=chr(ord('A') + dif % 26 - 1)
            else:
                result += 'Z'
        elif 97 <= ord(i) <= 122:
            dif = k - (ord('z') - ord(i))
            if dif % 26 != 0:
                result += chr(ord('a') + dif % 26 - 1)
            else:
                result += 'z'
        else:
            result += i
    return result
