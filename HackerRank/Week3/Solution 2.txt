#include <stdio.h>
#include <stdlib.h>

static void solve_one_case(void) {
    int n;
    if(scanf("%d", &n) != 1) return;
    int *q = (int *) malloc(n * sizeof(int));
    if (!q) return;
    
    for(int i = 0; i < n; ++i) scanf("%d", &q[i]);
    
    for(int i=0; i<n; i++) {
        if (q[i]-(i+1) > 2) {
            puts("Too chaotic");
            free(q);
            return;
        }
    }
    long long bribes = 0;
    for (int i=0; i<n; i++) {
        int start = q[i] - 2;
        if(start<0) start=0;
        for(int j=start; j<i; ++j) {
            if(q[j] > q[i]) ++bribes;
        }
    }
    
    printf("%lld\n", bribes);
    free(q);
}

int main(void) {
    int T;
    if(scanf("%d", &T) != 1) return 0;
    while(T--) solve_one_case(); 
    return 0;
}
