#include <stdio.h>
#include <string.h>
#include <math.h>
#include <stdlib.h>
#define MAX 100000

int stack1[MAX], stack2[MAX];
int top1 = -1, top2 = -1;

void push1(int x) { stack1[++top1] = x; }
int pop1() { return stack1[top1--]; }
int empty1() { return top1 == -1; }

void push2(int x) { stack2[++top2] = x; }
int pop2() { return stack2[top2--]; }
int empty2() { return top2 ==  -1; }

void enqueue(int x) {
    push1(x);
}

void dequeue() {
    if(empty2()) {
        while (!empty1()) {
            push2(pop1());
        }
    }
    if (!empty2()) {
        pop2();
    }
}

void printFront() {
    if (empty2()) {
        while (!empty1()) {
            push2(pop1());
        }
    }
    if (!empty2()) {
        printf("%d\n", stack2[top2]);
    }
}

int main() {
    /* Enter your code here. Read input from STDIN. Print output to STDOUT */     
    int q;
    scanf("%d", &q);
    
    while(q--) {
        int type, x;
        scanf("%d", &type);
        
        if(type == 1) {
            scanf("%d", &x);
            enqueue(x);   
        } else if (type == 2) {
            dequeue();
        } else if (type == 3) {
            printFront();
        }
    }
    return 0;
}
