#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// Huffman tree node
struct node {
    int freq;
    char data;
    struct node *left;
    struct node *right;
};

// Huffman decoding function
void decode_huff(struct node *root, char *s) {
    struct node *curr = root;
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == '0')
            curr = curr->left;
        else
            curr = curr->right;

        // If leaf node reached
        if (curr->left == NULL && curr->right == NULL) 
        {
            printf("%c", curr->data);
            curr = root; // reset back to root
        }
    }
}
