#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *a;
    int n, cap;
} MinHeap;

static void heap_init(MinHeap *h) {
    h->n = 0; h->cap = 16;
    h->a = (int*)malloc(h->cap * sizeof(int));
}

static void heap_free(MinHeap *h) {
    free(h->a); h->a = NULL; h->n = h->cap = 0;
}

static void heap_swap(int *x, int *y) { int t = *x; *x = *y; *y = t; }

static void heap_ensure(MinHeap *h) {
    if (h->n < h->cap) return;
    h->cap <<= 1;
    h->a = (int*)realloc(h->a, h->cap * sizeof(int));
}

static void heap_push(MinHeap *h, int v) {
    heap_ensure(h);
    int i = h->n++;
    h->a[i] = v;
    while (i > 0) {
        int p = (i - 1) >> 1;
        if (h->a[p] <= h->a[i]) break;
        heap_swap(&h->a[p], &h->a[i]);
        i = p;
    }
}
static int heap_empty(const MinHeap *h) { return h->n == 0; }

static int heap_top(const MinHeap *h) { return h->a[0]; }

static void heap_pop(MinHeap *h) {
    if (h->n == 0) return;
    h->a[0] = h->a[--h->n];
    int i = 0;
    while (1) {
        int l = i*2 + 1, r = l + 1, m = i;
        if (l < h->n && h->a[l] < h->a[m]) m = l;
        if (r < h->n && h->a[r] < h->a[m]) m = r;
        if (m == i) break;
        heap_swap(&h->a[i], &h->a[m]);
        i = m;
    }
}
static void sync_heaps(MinHeap *addH, MinHeap *delH) {
    while (!heap_empty(addH) && !heap_empty(delH) && heap_top(addH) == heap_top(delH)) {
        heap_pop(addH);
        heap_pop(delH);
    }
}

int main(void) {
    int q;
    if (scanf("%d", &q) != 1) return 0;

    MinHeap addH, delH;
    heap_init(&addH);
    heap_init(&delH);

    for (int i = 0; i < q; ++i) {
        int type; 
        scanf("%d", &type);
        if (type == 1) {
            int x; scanf("%d", &x);
            heap_push(&addH, x);
        } else if (type == 2) {
            int x; scanf("%d", &x);
            heap_push(&delH, x);
        } else if (type == 3) {
            sync_heaps(&addH, &delH);
            if (!heap_empty(&addH)) {
                printf("%d\n", heap_top(&addH));
            }
        }
    }
    heap_free(&addH);
    heap_free(&delH);
    return 0;
}
